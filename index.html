def calculate_bmi(weight_kg, height_cm): # แปลงส่วนสูงจาก เซนติเมตร เป็น เมตร height_m = height_cm / 100 # คำนวณค่า BMI bmi = weight_kg / (height_m**2) # แปลผลตามเกณฑ์มาตรฐาน (สำหรับคนเอเชีย) if bmi < 18.5: category = "น้ำหนักน้อย / ผอม" elif 18.5 <= bmi < 23.0: category = "ปกติ (สุขภาพดี)" elif 23.0 <= bmi < 25.0: category = "ท้วม / น้ำหนักเกิน" elif 25.0 <= bmi < 30.0: category = "อ้วนระดับ 1" else: category = "อ้วนระดับ 2 (อ้วนมาก)" return bmi, category # --- ตัวอย่างการรับค่าและแสดงผล --- try: weight = float(input("กรุณากรอกน้ำหนัก (กก.): ")) height = float(input("กรุณากรอกส่วนสูง (ซม.): ")) bmi_value, result = calculate_bmi(weight, height) print(f"\nค่า BMI ของคุณคือ: {bmi_value:.2f}") print(f"ผลการประเมิน: {result}") except ValueError: print("กรุณากรอกข้อมูลเป็นตัวเลขเท่านั้น") function calculateBMI(weightKg, heightCm) { const heightM = heightCm / 100; const bmi = weightKg / (heightM * heightM); let category = ""; if (bmi < 18.5) { category = "น้ำหนักน้อย / ผอม"; } else if (bmi < 23.0) { category = "ปกติ (สุขภาพดี)"; } else if (bmi < 25.0) { category = "ท้วม / น้ำหนักเกิน"; } else if (bmi < 30.0) { category = "อ้วนระดับ 1"; } else { category = "อ้วนระดับ 2 (อ้วนมาก)"; } return { bmi: bmi.toFixed(2), category: category }; } // ตัวอย่างการเรียกใช้งาน const result = calculateBMI(65, 170); // น้ำหนัก 65 kg, ส่วนสูง 170 cm console.log(`ค่า BMI: ${result.bmi}`); console.log(`ผลประเมิน: ${result.category}`);
